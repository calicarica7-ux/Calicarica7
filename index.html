MY PROJECT

Java Code

Cycling Guessing Game

A simple Java game where players guess which cyclist traveled farther.

5 Lives 5 Rounds Score System 
import java.util.Scanner;
import java.util.Random;

public class CyclingGuessGame {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int score = 0;
        int lives = 5;
        int rounds = 5;

        System.out.println("CYCLING GUESSING GAME");
        System.out.println("Guess who traveled farther!");
        System.out.println("You have 5 lives!");

        for (int i = 1; i <= rounds; i++) {
            if (lives == 0) {
                break;
            }

            int cyclistA = random.nextInt(50) + 1;
            int cyclistB = random.nextInt(50) + 1;

            System.out.println("ROUND " + i);
            System.out.println("Cyclist A: " + cyclistA + " km");
            System.out.println("Cyclist B: " + cyclistB + " km");

            if (cyclistA == cyclistB) {
                System.out.println("It's a tie!");
                continue;
            }

            System.out.print("Who traveled farther? (A/B): ");
            String guess = input.next().toUpperCase();

            if ((guess.equals("A") && cyclistA > cyclistB) ||
                (guess.equals("B") && cyclistB > cyclistA)) {
                System.out.println("Correct! +1 point!");
                score++;
            } else {
                System.out.println("Wrong guess!");
                lives--;
                System.out.println("You lost 1 life!");
            }

            System.out.println("Score: " + score);
            System.out.println("Lives: " + lives);
        }

        System.out.println("Final Score: " + score);
        System.out.println("Remaining Lives: " + lives);

        if (lives == 0) {
            System.out.println("GAME OVER!");
        } else {
            System.out.println("GAME FINISHED!");
        }

        if (score == 5) {
            System.out.println("Cycling Champion!");
        } else if (score >= 3) {
            System.out.println("Good job!");
        } else {
            System.out.println("Try again!");
        }
    }
}
